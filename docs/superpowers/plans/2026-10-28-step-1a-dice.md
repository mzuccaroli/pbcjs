# Step 1a — `packages/dice`: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Libreria `@pbcjs/dice` che trasforma una formula di tiro in AST e la valuta in modo sicuro e verificabile, con errori posizionali, limiti anti-abuso, `validate` per l'editor admin e una CLI di prova.

**Architecture:** Package solo sorgente come `@pbcjs/shared` (`exports: ./src/index.ts`). Pipeline in tre file: `lexer.ts` (stringa → token con posizione), `parser.ts` (token → AST, discesa ricorsiva), `evaluate.ts` (AST → `RollResult`). `validate.ts` usa il parser senza lanciare; `cli.ts` usa `roll`. Nessun `eval`; casualità iniettabile.

**Tech Stack:** TypeScript strict, Node 22 (`node:crypto`), Vitest 3, fast-check 4, tsx 4.

**Spec:** `docs/superpowers/specs/2026-10-28-step-1-dice-theme-design.md` §2, §4, §5, §6.

```ts
import { DiceError } from './errors'

export const MAX_FORMULA_LENGTH = 200

export type Punct = '+' | '-' | '*' | '/' | '(' | ')' | ','

export type Token =
  | { kind: 'int'; value: number; pos: number }
  | { kind: 'var'; name: string; pos: number }
  | { kind: 'punct'; value: Punct; pos: number }
  | { kind: 'd'; pos: number }
  | { kind: 'keep'; mode: 'kh' | 'kl'; pos: number }
  | { kind: 'fn'; name: 'min' | 'max'; pos: number }
  | { kind: 'eof'; pos: number }

const PUNCT = '+-*/(),'
const VAR_NAME = /^[a-z][a-z0-9_]*$/
const DIGIT = /[0-9]/
const LETTER = /[a-z]/i
const SPACE = /\s/

export function tokenize(formula: string): Token[] {
  if (formula.length > MAX_FORMULA_LENGTH) {
    throw new DiceError(
      'DICE_SYNTAX',
      `Formula is longer than ${MAX_FORMULA_LENGTH} characters`,
      MAX_FORMULA_LENGTH,
    )
  }

  const tokens: Token[] = []
  let i = 0
  while (i < formula.length) {
    const ch = formula.charAt(i)

    if (SPACE.test(ch)) {
      i++
    } else if (PUNCT.includes(ch)) {
      tokens.push({ kind: 'punct', value: ch as Punct, pos: i })
      i++
    } else if (DIGIT.test(ch)) {
      const start = i
      while (i < formula.length && DIGIT.test(formula.charAt(i))) i++
      const value = Number(formula.slice(start, i))
      if (!Number.isSafeInteger(value)) {
        throw new DiceError('DICE_SYNTAX', 'Number is too large', start)
      }
      tokens.push({ kind: 'int', value, pos: start })
    } else if (ch === '{') {
      const end = formula.indexOf('}', i)
      if (end === -1) throw new DiceError('DICE_SYNTAX', 'Unclosed variable', i)
      const name = formula.slice(i + 1, end)
      if (!VAR_NAME.test(name)) {
        throw new DiceError('DICE_SYNTAX', `Invalid variable name "${name}"`, i + 1)
      }
      tokens.push({ kind: 'var', name, pos: i })
      i = end + 1
    } else if (LETTER.test(ch)) {
      const start = i
      while (i < formula.length && LETTER.test(formula.charAt(i))) i++
      tokens.push(wordToken(formula.slice(start, i).toLowerCase(), start))
    } else {
      throw new DiceError('DICE_SYNTAX', `Unexpected character "${ch}"`, i)
    }
  }
  tokens.push({ kind: 'eof', pos: formula.length })
  return tokens
}

function wordToken(word: string, pos: number): Token {
  switch (word) {
    case 'd':
      return { kind: 'd', pos }
    case 'kh':
    case 'kl':
      return { kind: 'keep', mode: word, pos }
    case 'min':
    case 'max':
      return { kind: 'fn', name: word, pos }
    default:
      throw new DiceError('DICE_SYNTAX', `Unknown word "${word}"`, pos)
  }
}
```

Nota: `4d6kh3` viene letto come `4`, `d`, `6`, `kh`, `3` perché lettere e cifre sono sequenze separate. Il caso `' 4 d 6 kh 3 + {forza}'` del test funziona perché gli spazi sono ignorati.

```ts
import { DiceError } from './errors'
import { tokenize, type Punct, type Token } from './lexer'

export type DiceOperand =
  { type: 'num'; value: number } | { type: 'var'; name: string; pos: number }

export type DiceNode =
  | DiceOperand
  | {
      type: 'dice'
      count: DiceOperand
      sides: DiceOperand
      keep?: { mode: 'kh' | 'kl'; n: number }
    }
  | { type: 'neg'; expr: DiceNode }
  | { type: 'binary'; op: '+' | '-' | '*' | '/'; left: DiceNode; right: DiceNode }
  | { type: 'call'; fn: 'min' | 'max'; args: DiceNode[] }
  | { type: 'group'; expr: DiceNode }

export interface DiceAst {
  source: string
  root: DiceNode
}

export function parse(formula: string): DiceAst {
  const parser = new Parser(tokenize(formula))
  const root = parser.expr()
  parser.expectEnd()
  return { source: formula, root }
}

class Parser {
  private index = 0

  constructor(private readonly tokens: Token[]) {}

  expr(): DiceNode {
    let left = this.term()
    for (;;) {
      const token = this.peek()
      if (token.kind !== 'punct' || (token.value !== '+' && token.value !== '-')) return left
      this.next()
      left = { type: 'binary', op: token.value, left, right: this.term() }
    }
  }

  expectEnd(): void {
    if (this.peek().kind !== 'eof') throw this.error('Unexpected token')
  }

  private term(): DiceNode {
    let left = this.unary()
    for (;;) {
      const token = this.peek()
      if (token.kind !== 'punct' || (token.value !== '*' && token.value !== '/')) return left
      this.next()
      left = { type: 'binary', op: token.value, left, right: this.unary() }
    }
  }

  private unary(): DiceNode {
    if (this.isPunct('-')) {
      this.next()
      return { type: 'neg', expr: this.unary() }
    }
    return this.primary()
  }

  private primary(): DiceNode {
    const token = this.peek()
    switch (token.kind) {
      case 'int': {
        this.next()
        const operand: DiceOperand = { type: 'num', value: token.value }
        return this.peek().kind === 'd' ? this.dice(operand) : operand
      }
      case 'var': {
        this.next()
        const operand: DiceOperand = { type: 'var', name: token.name, pos: token.pos }
        return this.peek().kind === 'd' ? this.dice(operand) : operand
      }
      case 'd':
        return this.dice({ type: 'num', value: 1 })
      case 'fn':
        return this.call(token.name)
      case 'punct':
        if (token.value === '(') {
          this.next()
          const expr = this.expr()
          this.expectPunct(')')
          return { type: 'group', expr }
        }
        break
    }
    throw this.error('Expected a number, dice, variable, function or "("')
  }

  private dice(count: DiceOperand): DiceNode {
    this.next() // consume "d"
    const sidesToken = this.peek()
    let sides: DiceOperand
    if (sidesToken.kind === 'int') {
      sides = { type: 'num', value: sidesToken.value }
    } else if (sidesToken.kind === 'var') {
      sides = { type: 'var', name: sidesToken.name, pos: sidesToken.pos }
    } else {
      throw this.error('Expected the number of sides')
    }
    this.next()

    const keepToken = this.peek()
    if (keepToken.kind !== 'keep') return { type: 'dice', count, sides }
    this.next()
    const nToken = this.peek()
    if (nToken.kind !== 'int') throw this.error(`Expected a number after "${keepToken.mode}"`)
    this.next()
    return { type: 'dice', count, sides, keep: { mode: keepToken.mode, n: nToken.value } }
  }

  private call(fn: 'min' | 'max'): DiceNode {
    this.next() // consume function name
    this.expectPunct('(')
    const args = [this.expr()]
    while (this.isPunct(',')) {
      this.next()
      args.push(this.expr())
    }
    if (args.length < 2) throw this.error(`${fn}() needs at least 2 arguments`)
    this.expectPunct(')')
    return { type: 'call', fn, args }
  }

  private peek(): Token {
    return this.tokens[this.index] as Token
  }

  private next(): void {
    if (this.peek().kind !== 'eof') this.index++
  }

  private isPunct(value: Punct): boolean {
    const token = this.peek()
    return token.kind === 'punct' && token.value === value
  }

  private expectPunct(value: Punct): void {
    if (!this.isPunct(value)) throw this.error(`Expected "${value}"`)
    this.next()
  }

  private error(message: string): DiceError {
    const token = this.peek()
    const found = token.kind === 'eof' ? 'end of formula' : describe(token)
    return new DiceError('DICE_SYNTAX', `${message}, found ${found}`, token.pos)
  }
}

function describe(token: Token): string {
  switch (token.kind) {
    case 'int':
      return `"${token.value}"`
    case 'var':
      return `"{${token.name}}"`
    case 'punct':
      return `"${token.value}"`
    case 'd':
      return '"d"'
    case 'keep':
      return `"${token.mode}"`
    case 'fn':
      return `"${token.name}"`
    case 'eof':
      return 'end of formula'
  }
}
```

```ts
import type { Rng } from '../src/rng'

/** Returns the given values in order; throws when exhausted. */
export function sequenceRng(values: number[]): Rng {
  let index = 0
  return () => {
    const value = values[index++]
    if (value === undefined) throw new Error('sequenceRng exhausted')
    return value
  }
}

/** Deterministic mulberry32-based RNG for property tests. */
export function seededRng(seed: number): Rng {
  let state = seed >>> 0
  return (min, max) => {
    state = (state + 0x6d2b79f5) >>> 0
    let t = state
    t = Math.imul(t ^ (t >>> 15), t | 1)
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61)
    const r = ((t ^ (t >>> 14)) >>> 0) / 4294967296
    return min + Math.floor(r * (max - min + 1))
  }
}
```

```ts
import { DiceError } from './errors'
import { parse, type DiceAst, type DiceNode, type DiceOperand } from './parser'
import { cryptoRng, type Rng } from './rng'

export const MAX_DICE_PER_ROLL = 100
export const MIN_SIDES = 2
export const MAX_SIDES = 1000

export interface DieRoll {
  /** Resolved dice term, e.g. "4d6kh3". */
  die: string
  /** Every die rolled, in order. */
  results: number[]
  /** Indexes into results of the dice that count toward the total. */
  kept: number[]
}

export interface RollResult {
  total: number
  formula: string
  resolved: string
  rolls: DieRoll[]
}

export interface EvaluateOptions {
  vars?: Record<string, number>
  rng?: Rng
}

interface Context {
  vars: Record<string, number>
  rng: Rng
  diceUsed: number
  rolls: DieRoll[]
}

type VarNode = Extract<DiceOperand, { type: 'var' }>
type DiceTerm = Extract<DiceNode, { type: 'dice' }>

export function evaluate(ast: DiceAst, opts: EvaluateOptions = {}): RollResult {
  const ctx: Context = {
    vars: opts.vars ?? {},
    rng: opts.rng ?? cryptoRng,
    diceUsed: 0,
    rolls: [],
  }
  const total = evalNode(ast.root, ctx)
  return { total, formula: ast.source, resolved: render(ast.root, ctx.vars), rolls: ctx.rolls }
}

export function roll(formula: string, opts: EvaluateOptions = {}): RollResult {
  return evaluate(parse(formula), opts)
}

function evalNode(node: DiceNode, ctx: Context): number {
  switch (node.type) {
    case 'num':
      return node.value
    case 'var':
      return lookup(node, ctx.vars)
    case 'group':
      return evalNode(node.expr, ctx)
    case 'neg':
      return 0 - evalNode(node.expr, ctx)
    case 'call':
      return Math[node.fn](...node.args.map((arg) => evalNode(arg, ctx)))
    case 'dice':
      return evalDice(node, ctx)
    case 'binary': {
      const left = evalNode(node.left, ctx)
      const right = evalNode(node.right, ctx)
      switch (node.op) {
        case '+':
          return left + right
        case '-':
          return left - right
        case '*':
          return left * right
        case '/':
          if (right === 0) throw new DiceError('DICE_DIVISION_BY_ZERO', 'Division by zero')
          return Math.floor(left / right)
      }
    }
  }
}

function evalDice(node: DiceTerm, ctx: Context): number {
  const count = operandValue(node.count, ctx.vars)
  const sides = operandValue(node.sides, ctx.vars)
  if (!Number.isInteger(count) || count < 1) {
    throw new DiceError('DICE_LIMIT', `Dice count must be an integer of at least 1, got ${count}`)
  }
  if (!Number.isInteger(sides) || sides < MIN_SIDES || sides > MAX_SIDES) {
    throw new DiceError(
      'DICE_LIMIT',
      `Dice sides must be an integer between ${MIN_SIDES} and ${MAX_SIDES}, got ${sides}`,
    )
  }
  if (node.keep && (node.keep.n < 1 || node.keep.n > count)) {
    throw new DiceError('DICE_INVALID_KEEP', `Cannot keep ${node.keep.n} of ${count} dice`)
  }
  if (ctx.diceUsed + count > MAX_DICE_PER_ROLL) {
    throw new DiceError('DICE_LIMIT', `At most ${MAX_DICE_PER_ROLL} dice per roll`)
  }
  ctx.diceUsed += count

  const results = Array.from({ length: count }, () => ctx.rng(1, sides))
  const kept = node.keep
    ? keepIndexes(results, node.keep.mode, node.keep.n)
    : results.map((_, i) => i)
  const keepSuffix = node.keep ? `${node.keep.mode}${node.keep.n}` : ''
  ctx.rolls.push({ die: `${count}d${sides}${keepSuffix}`, results, kept })
  return kept.reduce((sum, i) => sum + (results[i] as number), 0)
}

/** Indexes of the n highest (kh) or lowest (kl) results; ties keep the lower index. */
function keepIndexes(results: number[], mode: 'kh' | 'kl', n: number): number[] {
  return results
    .map((_, i) => i)
    .sort((a, b) => {
      const diff = (results[b] as number) - (results[a] as number)
      return (mode === 'kh' ? diff : -diff) || a - b
    })
    .slice(0, n)
    .sort((a, b) => a - b)
}

function operandValue(operand: DiceOperand, vars: Record<string, number>): number {
  return operand.type === 'num' ? operand.value : lookup(operand, vars)
}

function lookup(node: VarNode, vars: Record<string, number>): number {
  const value = vars[node.name]
  if (!Object.hasOwn(vars, node.name) || value === undefined) {
    throw new DiceError('DICE_UNKNOWN_VARIABLE', `Unknown variable "{${node.name}}"`, node.pos)
  }
  return value
}

function render(node: DiceNode, vars: Record<string, number>): string {
  switch (node.type) {
    case 'num':
      return String(node.value)
    case 'var':
      return String(lookup(node, vars))
    case 'group':
      return `(${render(node.expr, vars)})`
    case 'neg':
      return `-${render(node.expr, vars)}`
    case 'binary':
      return `${render(node.left, vars)}${node.op}${render(node.right, vars)}`
    case 'call':
      return `${node.fn}(${node.args.map((arg) => render(arg, vars)).join(',')})`
    case 'dice': {
      const keepSuffix = node.keep ? `${node.keep.mode}${node.keep.n}` : ''
      return `${render(node.count, vars)}d${render(node.sides, vars)}${keepSuffix}`
    }
  }
}
```

```ts
import { DiceError } from './errors'
import { parse, type DiceAst, type DiceNode, type DiceOperand } from './parser'

export interface ValidateOptions {
  knownVars?: string[]
}

export interface ValidateResult {
  ok: boolean
  errors: DiceError[]
}

/** Checks syntax and, when knownVars is given, variable names. Never throws. */
export function validate(formula: string, opts: ValidateOptions = {}): ValidateResult {
  let ast: DiceAst
  try {
    ast = parse(formula)
  } catch (err) {
    if (err instanceof DiceError) return { ok: false, errors: [err] }
    throw err
  }

  const errors: DiceError[] = []
  if (opts.knownVars) {
    const known = new Set(opts.knownVars)
    for (const v of collectVars(ast.root)) {
      if (!known.has(v.name)) {
        errors.push(new DiceError('DICE_UNKNOWN_VARIABLE', `Unknown variable "{${v.name}}"`, v.pos))
      }
    }
  }
  return { ok: errors.length === 0, errors }
}

type VarNode = Extract<DiceOperand, { type: 'var' }>

function collectVars(node: DiceNode): VarNode[] {
  switch (node.type) {
    case 'num':
      return []
    case 'var':
      return [node]
    case 'group':
    case 'neg':
      return collectVars(node.expr)
    case 'binary':
      return [...collectVars(node.left), ...collectVars(node.right)]
    case 'call':
      return node.args.flatMap(collectVars)
    case 'dice':
      return [...collectVars(node.count), ...collectVars(node.sides)]
  }
}
```

```ts
import { DiceError } from './errors'
import { roll, type RollResult } from './evaluate'
import type { Rng } from './rng'

export interface CliIo {
  out(line: string): void
  err(line: string): void
}

const USAGE = 'Usage: roll "<formula>" [--var name=integer]... [--json]'
const VAR_ARG = /^([a-z][a-z0-9_]*)=(-?\d+)$/

export function runCli(argv: string[], io: CliIo, rng?: Rng): number {
  let formula: string | undefined
  let json = false
  const vars: Record<string, number> = {}

  for (let i = 0; i < argv.length; i++) {
    const arg = argv[i] as string
    if (arg === '--json') {
      json = true
    } else if (arg === '--var') {
      const spec = argv[++i] ?? ''
      const match = VAR_ARG.exec(spec)
      if (!match) return usageError(io, `Invalid --var "${spec}", expected name=integer`)
      vars[match[1] as string] = Number(match[2])
    } else if (formula === undefined) {
      formula = arg
    } else {
      return usageError(io, `Unexpected argument "${arg}"`)
    }
  }
  if (formula === undefined) return usageError(io, 'Missing formula')

  let result: RollResult
  try {
    result = roll(formula, { vars, rng })
  } catch (err) {
    if (!(err instanceof DiceError)) throw err
    io.err(`${err.code}: ${err.message}`)
    if (err.position !== undefined) {
      io.err(`  ${formula}`)
      io.err(`  ${' '.repeat(err.position)}^`)
    }
    return 1
  }

  if (json) {
    io.out(JSON.stringify(result, null, 2))
  } else {
    io.out(`total: ${result.total}`)
    io.out(`formula: ${result.formula}`)
    io.out(`resolved: ${result.resolved}`)
    for (const die of result.rolls) {
      const values = die.results.map((v, i) => (die.kept.includes(i) ? String(v) : `(${v})`))
      io.out(`${die.die}: [${values.join(', ')}]`)
    }
  }
  return 0
}

function usageError(io: CliIo, message: string): number {
  io.err(message)
  io.err(USAGE)
  return 2
}

const entry = process.argv[1]
if (entry && import.meta.url === pathToFileURL(entry).href) {
  process.exitCode = runCli(process.argv.slice(2), {
    out: (line) => console.log(line),
    err: (line) => console.error(line),
  })
}
```
